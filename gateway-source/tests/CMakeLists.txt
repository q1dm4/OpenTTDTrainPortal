cmake_minimum_required(VERSION 3.17)
project(gateway_pairing_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(pairing_tests pairing.cpp)
target_include_directories(pairing_tests PRIVATE ../engine)
# Assertions are the test oracle and must remain enabled in Release too.
if(MSVC)
    target_compile_options(pairing_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(pairing_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME pairing_policy COMMAND pairing_tests)
